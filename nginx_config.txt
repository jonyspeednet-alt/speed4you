server {
    listen 443 ssl http2;
    server_name data.speed4you.net;
    ssl_certificate /home/speed4you/.acme.sh/data.speed4you.net/fullchain.cer;
    ssl_certificate_key /home/speed4you/.acme.sh/data.speed4you.net/data.speed4you.net.key;
    root /var/www/html;
    index index.php index.html;
    access_log /var/www/html/Extra_Storage/nginx-logs/data-access.log;
    error_log /var/www/html/Extra_Storage/nginx-logs/data-error.log;
    proxy_temp_path /var/www/html/Extra_Storage/nginx-tmp/proxy 1 2;
    sendfile on;
    tcp_nopush on;
    open_file_cache max=1000 inactive=20s;
    open_file_cache_valid 30s;
    open_file_cache_min_uses 2;
    open_file_cache_errors on;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;
    keepalive_timeout 65;
    keepalive_requests 200;
    gzip on;
    gzip_types text/plain text/css application/json application/javascript image/svg+xml;
    gzip_min_length 500;
    gzip_comp_level 4;
    gzip_proxied any;
    gzip_vary on;
    location /.well-known/acme-challenge/ {
        root /var/www/html;
    }
    location / {
        try_files $uri /_h5ai/public/index.php?$args;
    }
    location ~ \.php$ {
        fastcgi_pass unix:/run/php/php8.3-fpm.sock;
        fastcgi_index index.php;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_buffer_size 128k;
        fastcgi_buffers 256 16k;
        fastcgi_busy_buffers_size 256k;
        fastcgi_cache H5AI_CACHE;
        fastcgi_cache_key "$scheme$request_method$host$request_uri";
        fastcgi_cache_valid 200 5m;
        fastcgi_cache_use_stale updating error timeout invalid_header;
        fastcgi_cache_lock on;
        add_header X-FastCGI-Cache $upstream_cache_status;
    }
    location ~* \.(css|js|png|jpg|jpeg|gif|ico|svg|woff2?|ttf|eot)$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }
}
